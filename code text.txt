        // Render Functions
        function renderGrid() {
            const grid = document.getElementById('communitiesGrid');
            grid.innerHTML = '';

            const searchVal = document.getElementById('searchInput').value.toLowerCase();
            const councilVal = document.getElementById('councilFilter').value;
            const typeVal = document.getElementById('typeFilter').value;
            const statusVal = document.getElementById('statusFilter').value;

            const filtered = state.communities.filter(c => {
                // If not admin, only show approved ones
                if (!state.isAdmin && !c.approved) return false;

                const matchesSearch = c.name.toLowerCase().includes(searchVal) || c.councilName.includes(searchVal);
                const matchesCouncil = councilVal === 'all' || c.council === councilVal;
                const matchesType = typeVal === 'all' || c.type === typeVal;
                const matchesStatus = statusVal === 'all' || c.status === statusVal;

                return matchesSearch && matchesCouncil && matchesType && matchesStatus;
            });

            if (filtered.length === 0) {
                grid.innerHTML = `
                    <div class="col-span-full py-12 text-center text-slate-500">
                        <i class="fa-solid fa-folder-open text-4xl mb-3"></i>
                        <p class="lang-he">לא נמצאו יישובים התואמים את הסינון.</p>
                        <p class="lang-en hidden">No communities match your filter criteria.</p>
                    </div>

                return;

                let statusBg = c.status === 'open' ? 'bg-emerald-500/10 text-emerald-400 border-emerald-500/30' :

        function handleFormSubmit(e) {
            e.preventDefault();
            alert(state.currentLang === 'he' ? 'הדיווח נשלח בהצלחה וממתין לאישור מנהל המערכת.' : 'Submitted successfully! Awaiting admin review.');
            closeSubmitModal();

        // Admin Actions
        function toggleAdminMode() {
            state.isAdmin = !state.isAdmin;
            const btnHe = document.getElementById('adminBtnText');
            const btnEn = document.getElementById('adminBtnTextEn');
            const noticeBox = document.getElementById('adminNoticeBox');

            if (state.isAdmin) {
                btnHe.innerText = 'מצב מנהל (פעיל)';
                btnEn.innerText = 'Admin Mode (On)';
                noticeBox.classList.remove('hidden');
            } else {
                btnHe.innerText = 'מצב מנהל (כבוי)';
                btnEn.innerText = 'Admin Mode (Off)';
                noticeBox.classList.add('hidden');
            }
            renderGrid();
        }

        function approveCommunity(id) {
            const item = state.communities.find(c => c.id === id);
            if (item) {
                item.approved = true;
                saveToStorage();
                renderGrid();
            }
        }

        function rejectCommunity(id) {
            state.communities = state.communities.filter(c => c.id !== id);
            saveToStorage();
            renderGrid();
        }

        // Language Toggle Logic
        function toggleLanguage() {
            state.currentLang = state.currentLang === 'he' ? 'en' : 'he';
            document.documentElement.dir = state.currentLang === 'he' ? 'rtl' : 'ltr';
            document.getElementById('langSwitchLabel').innerText = state.currentLang === 'he' ? 'EN' : 'HE';
            applyLanguageUI();
        }

        function applyLanguageUI() {
            const isHe = state.currentLang === 'he';
            document.querySelectorAll('.lang-he').forEach(el => {
                if (isHe) el.classList.remove('hidden'); else el.classList.add('hidden');
            });
            document.querySelectorAll('.lang-en').forEach(el => {
                if (!isHe) el.classList.remove('hidden'); else el.classList.add('hidden');
            });
        }
